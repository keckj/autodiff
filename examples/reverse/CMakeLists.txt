file(GLOB_RECURSE CUFILES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} *.cu)

foreach(CUFILE ${CUFILES})
    get_filename_component(FILENAME ${CUFILE} NAME_WE)
    add_executable(${FILENAME} ${CUFILE})
    target_link_libraries(${FILENAME} PRIVATE autodiff::autodiff CUDA::nvrtc)
    if (USE_SYSTEM_EIGEN)
        target_link_libraries(${FILENAME} PRIVATE Eigen3::Eigen)
    endif()
    set_target_properties(${FILENAME} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endforeach()
